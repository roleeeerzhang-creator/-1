云端同步设置

配置 Supabase 后，所有手动添加的节点将存储在云端，团队成员共享同一份数据。

如何获取 Supabase 配置？
  1. 前往 supabase.com 免费注册并创建一个 Project
  2. 在 Project Settings → API 中复制 Project URL 和 anon public key
  3. 在 SQL Editor 中执行下方建表语句，然后回到此处填写并保存
create table if not exists calendar_events (
  id uuid default gen_random_uuid() primary key,
  date text not null,
  type text not null,
  title text not null,
  note text,
  time text,
  remind text,
  done boolean default false,
  created_at timestamptz default now()
);
alter table calendar_events enable row level security;
create policy "public access" on calendar_events
  for all using (true) with check (true);
